# The scenario .github/workflows/agent-drive.yml runs when none is given: one pass over every kind of
# step, on Scripts/Fixtures/smoke-rich.md, so a change to the driver shows that each still works.
# Steps are listed at the top of Scripts/agent-drive.swift.
shot opened
tree opened
focused

# A real click into the first paragraph, a caret placed through accessibility at the start of the
# heading, then real keys: one, then the rest, with the caret read between them.
clickat 300 160
focused
caret "Markdown WYSIWYG" 0
type "A"
wait 1
selection "Markdown WYSIWYG"
type "gent "
wait 1
selection "Markdown WYSIWYG"
shot typed
expect-value "Markdown WYSIWYG" "Agent Typography proof"

# Formatting from the toolbar, then from the keyboard (through the menu bar's key equivalents),
# and cmd-S saves it.
caret "Markdown WYSIWYG" 0 5
press Bold
wait 1
shot bold
caret "Markdown WYSIWYG" 6 10
menu-items Format
key cmd+i
wait 1
shot italic
key cmd+s
wait 2
file saved
expect-file "**Agent**"
expect-file "*Typography*"

# The mode control says which mode the editor is in, however the mode was chosen: the toolbar,
# the View menu, the keyboard.
press Source
wait 1
shot source
expect-value Source 1
expect-value WYSIWYG 0
value "Markdown source"
menu-items View
menu View "WYSIWYG Mode"
wait 1
shot wysiwyg-from-menu
expect-value WYSIWYG 1
# From WYSIWYG whatever the menu did, so the shortcut's check cannot pass by standing still.
press WYSIWYG
wait 1
key ctrl+cmd+2
wait 1
shot source-from-keyboard
